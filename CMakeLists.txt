cmake_minimum_required(VERSION 3.25)
project(BrettspielRegal VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BRETTSPIELREGAL_SANITIZERS "Enable address and undefined behavior checks" OFF)
include(CTest)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Sql)
if(BUILD_TESTING)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
endif()

add_library(project_options INTERFACE)
if(MSVC)
  target_compile_options(project_options INTERFACE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
  if(BRETTSPIELREGAL_SANITIZERS)
    target_compile_options(project_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(project_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()

add_library(game_domain src/domain/game.cpp src/application/game_service.cpp)
target_include_directories(game_domain PUBLIC src)
target_link_libraries(game_domain PUBLIC Qt6::Core PRIVATE project_options)
add_library(game_storage src/persistence/sqlite_game_repository.cpp src/persistence/data_directory.cpp)
target_link_libraries(game_storage PUBLIC game_domain Qt6::Sql PRIVATE project_options)
add_library(game_ui src/ui/storage_controller.cpp src/ui/game_editor.cpp src/ui/main_window.cpp src/ui/ui_support.cpp)
target_link_libraries(game_ui PUBLIC game_domain Qt6::Widgets PRIVATE project_options)
qt_add_executable(brettspielregal WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(brettspielregal PRIVATE game_storage game_ui project_options)
target_compile_definitions(brettspielregal PRIVATE APP_VERSION="${PROJECT_VERSION}")
qt_add_resources(brettspielregal notices PREFIX "/licenses" FILES THIRD_PARTY_NOTICES.md)
qt_add_resources(brettspielregal branding PREFIX "/" FILES assets/branding/icon-256.png)
if(WIN32)
  enable_language(RC)
  configure_file(packaging/windows/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
  target_sources(brettspielregal PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()
set_target_properties(brettspielregal PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER de.appfabrik.brettspielregal
  MACOSX_BUNDLE_BUNDLE_NAME BrettspielRegal
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist.in")

if(BUILD_TESTING)
  foreach(test_name domain repository dialogs)
    add_executable(${test_name}_tests tests/${test_name}_test.cpp)
    target_link_libraries(${test_name}_tests PRIVATE game_storage game_ui Qt6::Test project_options)
    add_test(NAME ${test_name} COMMAND ${test_name}_tests)
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endforeach()
endif()

install(TARGETS brettspielregal BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION licenses)
install(DIRECTORY licenses/ DESTINATION licenses)
if(WIN32 OR APPLE)
  qt_generate_deploy_app_script(TARGET brettspielregal OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
